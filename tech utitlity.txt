* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Static Public Method Z2CL_TECH_COMM_UTILITIES_CLD=>CONV_WEIGHT
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_UNIT                        TYPE        CL_UOM_CONVERSION=>TY_MSEHI
* | [--->] IV_WEIGHT                      TYPE        I_SALESDOCUMENTITEM-ITEMNETWEIGHT
* | [--->] IV_TRG_UNIT                    TYPE        CL_UOM_CONVERSION=>TY_MSEHI (default ='LB')
* | [<-()] RV_WEIGHT                      TYPE        I_SALESDOCUMENTITEM-ITEMNETWEIGHT
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD conv_weight.
*---------------------------------------------------------------------------*

*  Class Name            : ZCL_TECH_COMM_UTILITIES_CLD
*  Method name           : CONV_WEIGHT
*  Created By            : Parthiban Dhamodharan
*  Created On            : 11/08/2025
*  Purpose               : Method for simple conversion of weight from source
*                          to target Unit
*  RICEFW ID-Description : Common Utility Build
*----------------------------------------------------------------------------*
* MODIFICATION LOG                                                           *
*----------------------------------------------------------------------------*
*Date            Dev ID     TR#          Description                         *
*----------------------------------------------------------------------------*
*DD-MMM-YYYY    SAPID      XXXXXXXXX     Change description                  *
******************************************************************************

    DATA: lv_result  TYPE i_salesdocumentitem-itemnetweight . " total weight

    CLEAR: rv_weight.
*If source & target unit is same, return the same weight
    IF iv_unit = iv_trg_unit.
      rv_weight = iv_weight.
      RETURN.
    ENDIF.
*Convert Weight from source to target Unit
    DATA(lo_unit) = cl_uom_conversion=>create( ).

    IF lo_unit IS BOUND.
      "convert weight
      lo_unit->unit_conversion_simple( EXPORTING  input                = iv_weight
                                                  round_sign           = abap_false
                                                  unit_in              = iv_unit
                                                  unit_out             = iv_trg_unit
                                       IMPORTING  output               = lv_result
                                       EXCEPTIONS conversion_not_found = 01
                                                  division_by_zero     = 02
                                                  input_invalid        = 03
                                                  output_invalid       = 04
                                                  overflow             = 05
                                                  units_missing        = 06
                                                  unit_in_not_found    = 07
                                                  unit_out_not_found   = 08 ).
      IF sy-subrc = 0.
*Pass the converted  weight as result
        rv_weight = lv_result.
      ENDIF.
    ENDIF.
  ENDMETHOD.

  METHOD generate_log.
*---------------------------------------------------------------------*

*  Class Name            : ZCL_TECH_COMM_UTILITIES_CLD
*  Method name           : GENERATE_LOG
*  Created By            : Parthiban Dhamodharan
*  Created On            : 11/08/2025

*  RICEFW ID-Description : Common Utility Class Build

* MODIFICATION LOG                                                     *

* Date            Dev ID     TR#          Description                  *
*----------------------------------------------------------------------*
* DD-MMM-YYYY    SAPID        XXXXXXXXX   Change description           *

    CLEAR : eo_lo_log.
    TRY.
        " Create and set header
        DATA(lo_header) = cl_bali_header_setter=>create( object      = iv_object
                                                         subobject   = iv_subobject
                                                         external_id = cl_system_uuid=>create_uuid_c32_static( )
                          )->set_expiry( expiry_date       = CONV d( cl_abap_context_info=>get_system_date( ) + 30 )
                                         keep_until_expiry = abap_true ).

        IF lo_header IS BOUND.
          " Create instance of log object
          eo_lo_log = cl_bali_log=>create( ).
          IF eo_lo_log IS BOUND.
            " set header to log object instance
            eo_lo_log->set_header( lo_header ).
          ENDIF.
        ENDIF.
      CATCH cx_bali_runtime.
        ASSERT 1 = 2.
      CATCH cx_uuid_error.
        ASSERT 1 = 2.
    ENDTRY.
  ENDMETHOD.

  METHOD get_enhancement_switch.
*---------------------------------------------------------------------*

*  Class Name            : ZCL_TECH_COMM_UTILITIES_CLD
*  Method name           : GET_ENHANCEMENT_SWITCH
*  Created By            : Parthiban Dhamodharan
*  Created On            : 11/08/2025
*  Purpose               : Method to get enhancement switch framework
*  RICEFW ID-Description : Common Utility Class Build
*----------------------------------------------------------------------*
* MODIFICATION LOG                                                     *
*----------------------------------------------------------------------*
*Date           Dev ID     TR#          Description                    *
*----------------------------------------------------------------------*
*DD-MMM-YYYY    SAPID      XXXXXXXXX   Change description              *
************************************************************************
*Clear the returning parameter before the call
    CLEAR rv_enh_switch.
*Fetch the switch from Custom CDS
    SELECT zmodule,
           zricefwid,
           zcounter,
           zwitchind
     FROM  z2dev_if_actvt_deact_enhn_bcm
    WHERE  zmodule = @iv_process_area
      AND  zricefwid = @iv_ricefw_id
    ORDER BY zcounter
    INTO TABLE @DATA(lt_enh_switch).
    IF sy-subrc = 0.
      DATA(ls_enh_switch) = VALUE #( lt_enh_switch[ zcounter  = iv_counter ] OPTIONAL ).
      IF ls_enh_switch IS NOT INITIAL.
** If the switch is active then pass this to the returning parameter which can be used to check
** whether the enhancement should get trigger or not.
        rv_enh_switch = ls_enh_switch-zwitchind.
      ENDIF.
    ENDIF.
  ENDMETHOD.

  METHOD get_partner_addr.
*---------------------------------------------------------------------*

*  Class Name            : ZCL_TECH_COMM_UTILITIES_CLD
*  Method name           : GET_PARTNER_ADDR
*  Created By            : Parthiban Dhamodharan
*  Created On            : 11/08/2025
*  Purpose               : Method to fetch partner details for different
*                          partner functions including address details
*  RICEFW ID-Description : Common Utility Class Build
*----------------------------------------------------------------------*
* MODIFICATION LOG                                                     *
*----------------------------------------------------------------------*
*Date           Dev ID     TR#          Description                    *
*----------------------------------------------------------------------*
*DD-MMM-YYYY    SAPID      XXXXXXXXX   Change description              *
************************************************************************
    CONSTANTS:
      lc_parvw_sp TYPE i_salesdocumentpartner-partnerfunction  VALUE 'AG',
      lc_parvw_bp TYPE i_salesdocumentpartner-partnerfunction  VALUE 'BP',
      lc_parvw_sh TYPE i_salesdocumentpartner-partnerfunction  VALUE 'WE',
      lc_parvw_py TYPE i_salesdocumentpartner-partnerfunction  VALUE 'PY'.

    IF iv_vbeln IS NOT INITIAL AND it_adrnr IS INITIAL.
      "Get address number based on Sales order
      SELECT a~customer AS kunnr,
             a~addressid AS adrnr,
             a~partnerfunction AS parvw
      FROM i_salesdocumentpartner AS a
      WHERE salesdocument =  @iv_vbeln
        INTO TABLE @DATA(lt_adrc).
      IF sy-subrc NE 0.
        RETURN.
      ELSE.
        " map obtained address number based on sales order number
        DATA(lr_customer) = VALUE gr_customer( FOR ls_add IN lt_adrc ( sign = gc_sign_include option = gc_opt_equal low = ls_add-kunnr ) ).
      ENDIF.
    ENDIF.
*Get customer address details based on address number
    IF it_adrnr[] IS NOT INITIAL.

      SELECT customer,
             addressid,
             customername,
             streetname,
             cityname,
             country,
             region,
             telephonenumber1,
             postalcode
        FROM i_customer WITH PRIVILEGED ACCESS
        WHERE addressid IN @it_adrnr
        ORDER BY customer,addressid
        INTO TABLE @DATA(lt_cust).
      IF sy-subrc <> 0.
        CLEAR lt_cust.
      ENDIF.

    ELSEIF lr_customer[] IS NOT INITIAL.

      SELECT customer,
             addressid,
             customername,
             streetname,
             cityname,
             country,
             region,
             telephonenumber1,
             postalcode
        FROM i_customer WITH PRIVILEGED ACCESS
        WHERE customer IN @lr_customer
        ORDER BY customer,addressid
        INTO TABLE @lt_cust.
      IF sy-subrc <> 0.
        CLEAR lt_cust.
      ENDIF.
    ENDIF.

    IF lt_cust IS NOT INITIAL.
      "-- Get Country Name details
      SELECT DISTINCT a~language,
                      a~country AS land1,
                      a~countryname AS landx,
                      a~nationalityname
       FROM i_countrytext AS a
       INNER JOIN @lt_cust AS b
       ON a~country = b~country AND
          a~language = @sy-langu
       INTO TABLE @DATA(lt_t005t).

      IF sy-subrc IS NOT INITIAL.
        CLEAR: lt_t005t[].
      ENDIF.
    ENDIF.
*----**Sold to Customer details and Address
    DATA(ls_adrc) = VALUE #( lt_adrc[ parvw = lc_parvw_sp ] OPTIONAL ).
    IF ls_adrc IS NOT INITIAL.
      DATA(ls_cust) = VALUE #( lt_cust[ customer = ls_adrc-kunnr ] OPTIONAL ).
      rs_addr-cust_number     =  |{ ls_adrc-kunnr ALPHA = OUT }|.
      rs_addr-cust_street     =  ls_cust-streetname.
      rs_addr-cust_city1      =  ls_cust-cityname.
      rs_addr-cust_region     =  ls_cust-region.
      rs_addr-cust_country    =  ls_cust-country.
      rs_addr-cust_countryx   = VALUE #( lt_t005t[ land1 = ls_cust-country ]-landx OPTIONAL ).
      rs_addr-cust_name1      = ls_cust-customername.
      rs_addr-cust_tel_number = ls_cust-telephonenumber1.
      rs_addr-cust_po_code    = ls_cust-postalcode.
    ENDIF.
    CLEAR: ls_adrc, ls_cust.
*----**Ship To details and Address
    ls_adrc = VALUE #( lt_adrc[ parvw = lc_parvw_sh ] OPTIONAL ).
    IF ls_adrc IS NOT INITIAL.
      ls_cust = VALUE #( lt_cust[ customer = ls_adrc-kunnr ] OPTIONAL ).
      rs_addr-shp_number     = |{ ls_adrc-kunnr ALPHA = OUT }|.
      rs_addr-shp_street     =  ls_cust-streetname.
      rs_addr-shp_city1      =  ls_cust-cityname.
      rs_addr-shp_region     =  ls_cust-region.
      rs_addr-shp_country    =  ls_cust-country.
      rs_addr-shp_countryx   = VALUE #( lt_t005t[ land1 = ls_cust-country ]-landx OPTIONAL ).
      rs_addr-shp_name1      = ls_cust-customername.
      rs_addr-shp_tel_number = ls_cust-telephonenumber1.
      rs_addr-shp_po_code    = ls_cust-postalcode.
    ENDIF.
    CLEAR: ls_adrc, ls_cust.
*----**Bill To details and Address
    ls_adrc = VALUE #( lt_cust[ customer = ls_adrc-kunnr ] OPTIONAL ).
    IF ls_adrc IS NOT INITIAL.
      ls_cust = VALUE #( lt_cust[ addressid = ls_adrc-adrnr ] OPTIONAL ).
      rs_addr-bil_number     = |{ ls_adrc-kunnr ALPHA = OUT }|.
      rs_addr-bil_street     =  ls_cust-streetname.
      rs_addr-bil_city1      =  ls_cust-cityname.
      rs_addr-bil_region     =  ls_cust-region.
      rs_addr-bil_country    =  ls_cust-country.
      rs_addr-bil_countryx   = VALUE #( lt_t005t[ land1 = ls_cust-country ]-landx OPTIONAL ).
      rs_addr-bil_name1      = ls_cust-customername.
      rs_addr-bil_tel_number = ls_cust-telephonenumber1.
      rs_addr-bil_po_code    = ls_cust-postalcode.
    ENDIF.
    CLEAR: ls_adrc, ls_cust.
*----**Payer details and Address
    ls_adrc = VALUE #( lt_cust[ customer = ls_adrc-kunnr ] OPTIONAL ).
    IF ls_adrc IS NOT INITIAL.
      ls_cust = VALUE #( lt_cust[ addressid = ls_adrc-adrnr ] OPTIONAL ).
      rs_addr-pay_number     = |{ ls_adrc-kunnr ALPHA = OUT }|.
      rs_addr-pay_street     =  ls_cust-streetname.
      rs_addr-pay_city1      =  ls_cust-cityname.
      rs_addr-pay_region     =  ls_cust-region.
      rs_addr-pay_country    =  ls_cust-country.
      rs_addr-pay_countryx   = VALUE #( lt_t005t[ land1 = ls_cust-country ]-landx OPTIONAL ).
      rs_addr-pay_name1      = ls_cust-customername.
      rs_addr-pay_tel_number = ls_cust-telephonenumber1.
      rs_addr-pay_po_code    = ls_cust-postalcode.

  METHOD send_email.
*---------------------------------------------------------------------*

*  Class Name            : ZCL_TECH_COMM_UTILITIES_CLD
*  Method name           : PDF_EMAIL
*  Created By            : Parthiban Dhamodharan
*  Created On            : 11/08/2025
*  Purpose               : Method to send PDF attachement in email
*  RICEFW ID-Description : Common Utility Class Build
*----------------------------------------------------------------------*
* MODIFICATION LOG                                                     *
*----------------------------------------------------------------------*
* Date           Dev ID     TR#          Description                   *
*----------------------------------------------------------------------*
* DD-MMM-YYYY    SAPID     XXXXXXXXX    Change description             *
************************************************************************
    DATA: lv_recipient TYPE cl_bcs_mail_message=>ty_address,
          lo_mail      TYPE REF TO cl_bcs_mail_message.

    CONSTANTS : lc_content_pdf   TYPE cl_bcs_mail_textpart=>ty_content_type VALUE 'application/pdf', "PDF content
                lc_content_excel TYPE cl_bcs_mail_textpart=>ty_content_type VALUE 'application/xlsx', "Excel content
                lc_content_txt   TYPE cl_bcs_mail_textpart=>ty_content_type VALUE 'application/txt'. "Text content

    CLEAR : et_status[].
    TRY.
        " create instance of CL_BCS_MAIL_MESSAGE
        lo_mail = cl_bcs_mail_message=>create_instance( ).

        IF lo_mail IS BOUND.
*Set Email subject
          lo_mail->set_subject( iv_subject ).
*Set Email Body
*Plain text email body
          IF iv_html_body IS INITIAL.
            lo_mail->set_main( cl_bcs_mail_textpart=>create_instance( iv_content      = iv_body
                                                                      iv_content_type = 'text/plain' ) ).
*HTML text email body
          ELSE.
            lo_mail->set_main( cl_bcs_mail_textpart=>create_instance( iv_content      = iv_body
                                                                      iv_content_type = 'text/html' ) ).
*Set Email sender
          ENDIF.
          lo_mail->set_sender( iv_sender ).

*Add Recipient Email IDs
          IF ct_email[] IS NOT INITIAL.
            LOOP AT ct_email ASSIGNING FIELD-SYMBOL(<lfs_email>).
              lv_recipient = <lfs_email>-smtp_addr.
              lo_mail->add_recipient( iv_address = lv_recipient
                                      iv_copy    = cl_bcs_mail_message=>to ).
              CLEAR lv_recipient.
            ENDLOOP.
          ENDIF.

*Add CC Recipient Email IDs
          IF ct_email_cc[] IS NOT INITIAL.
            LOOP AT ct_email_cc ASSIGNING FIELD-SYMBOL(<lfs_email_cc>).
              lv_recipient = <lfs_email_cc>-smtp_addr.
              lo_mail->add_recipient( iv_address = lv_recipient
                                      iv_copy    = cl_bcs_mail_message=>cc ).
              CLEAR lv_recipient.
            ENDLOOP.
          ENDIF.

*Add BCC Recipient Email IDs
          IF ct_email_bcc[] IS NOT INITIAL.
            LOOP AT ct_email_bcc ASSIGNING FIELD-SYMBOL(<lfs_email_bcc>).
              lv_recipient = <lfs_email_bcc>-smtp_addr.
              lo_mail->add_recipient( iv_address = lv_recipient
                                      iv_copy    = cl_bcs_mail_message=>bcc ).
              CLEAR lv_recipient.
            ENDLOOP.
          ENDIF.
** Add attachment
          IF it_attachment IS NOT INITIAL.
            LOOP AT it_attachment INTO DATA(ls_attachment).
              lo_mail->add_attachment( cl_bcs_mail_binarypart=>create_instance(
                     iv_content      = ls_attachment-content
                     iv_content_type =  COND #( WHEN ls_attachment-file_type = 'X' THEN lc_content_excel " Excel content
                                                WHEN ls_attachment-file_type = 'P' THEN lc_content_pdf   " PDF content
                                                WHEN ls_attachment-file_type = 'T' THEN lc_content_txt   " TXT content
                                                ELSE lc_content_pdf ) "Default is PDF content
                     iv_filename     = ls_attachment-file_name
                   ) ).
            ENDLOOP.
          ELSEIF iv_attachment IS NOT INITIAL.
            lo_mail->add_attachment( cl_bcs_mail_binarypart=>create_instance(
             iv_content      = iv_attachment
             iv_content_type =  COND #( WHEN iv_content_type = 'X' THEN lc_content_excel " Excel content
                                        WHEN iv_content_type = 'P' THEN lc_content_pdf   " PDF content
                                        WHEN iv_content_type = 'T' THEN lc_content_txt   " TXT content
                                        ELSE lc_content_pdf ) "Default is PDF content
             iv_filename     = iv_filename
           ) ).
          ENDIF.
          IF iv_async_mode IS INITIAL.
*Send email
            lo_mail->send(
              IMPORTING
                et_status      = et_status
                ev_mail_status = DATA(lv_mail_status)
            ).
*Send email in asynchronous mode where explicit commit work is needed
          ELSE.
            TRY.
                lo_mail->send_async( ).
              CATCH cx_bcs_mail INTO DATA(lx_mail).
                DATA(lv_msg) = lx_mail->get_text( ).
            ENDTRY.
          ENDIF.
*-   Addded to clear EPC as lv_mail_status is not in use.
          IF NOT lv_mail_status IS INITIAL.
            CLEAR: lv_mail_status.
          ENDIF.
        ENDIF.
      CATCH cx_bcs_mail INTO DATA(lx_bcs_mail).
        DATA(lv_errormsg) = lx_bcs_mail->get_text( ).
    ENDTRY.

  ENDMETHOD.

* | Static Public Method Z2CL_TECH_COMM_UTILITIES_CLD=>SPLIT_DATA_FROM_FILE
* +-------------------------------------------------------------------------------------------------+

* | [--->] IV_SEPARATOR                   TYPE        ABAP_CHAR1
* | [<-->] CT_DATA_TABLE                  TYPE        TABLE
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD split_data_from_file.
*---------------------------------------------------------------------*
*  Class Name           : ZCL_TECH_COMM_UTILITIES_CLD
*  Method name          : SPLIT_DATA_FROM_FILE
*  Created By           : Parthiban Dhamodharan
*  Created On           : 11/08/2025
*  Purpose              : Method to split data from AL11 file
*  RICEFWID-Description : Common Utility Build
*----------------------------------------------------------------------*
* MODIFICATION LOG                                                     *
*----------------------------------------------------------------------*
*Date            Dev ID     TR#          Description                   *
*----------------------------------------------------------------------*
*DD-MMM-YYYY    SAPID       XXXXXXXXX   Change description             *
************************************************************************
* Split record at delimiter and append to data table
    DATA: lt_values TYPE STANDARD TABLE OF string,
          lo_ref    TYPE REF TO data.

    FIELD-SYMBOLS: <lfs_i> TYPE any,
                   <lfs_v> TYPE any.

* Create structure like line of INPUT TABLE
    CREATE DATA lo_ref LIKE LINE OF ct_data_table.
    ASSIGN lo_ref->* TO <lfs_i>.
* Split at the separator
    SPLIT iv_data AT iv_separator INTO TABLE lt_values.
    IF sy-subrc EQ 0.
      DATA(lv_lines) = lines( lt_values ).
      DATA(lv_sep) = cl_abap_char_utilities=>cr_lf+0(1).
      LOOP AT lt_values INTO DATA(ls_value).
        IF sy-tabix = lv_lines.
          IF ls_value CS lv_sep.
            REPLACE lv_sep IN ls_value WITH space.
          ENDIF.
        ENDIF.
        ASSIGN COMPONENT sy-tabix OF STRUCTURE <lfs_i> TO <lfs_v>.
        IF sy-subrc = 0 AND <lfs_v> IS ASSIGNED.
*Add the record to internal table
          CONDENSE ls_value.
          <lfs_v> = ls_value.
        ENDIF.
      ENDLOOP.
      APPEND <lfs_i> TO ct_data_table.
    ENDIF.
  ENDMETHOD.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Static Public Method Z2CL_TECH_COMM_UTILITIES_CLD=>SPLIT_FILE_NAME
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_FILENAME                    TYPE        STRING(optional)
* | [--->] IV_LOCATION                    TYPE        ABAP_CHAR1(optional)
* | [<---] EV_PATH                        TYPE        STRING
* | [<---] EV_NAME                        TYPE        STRING
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD split_file_name.
*---------------------------------------------------------------------*
*  Class Name            : ZZCL_TECH_COMM_UTILITIES_CLD
*  Method name           : SPLIT_FILE_NAME
*  Created By            : Parthiban Dhamodharan
*  Created On            : 11/08/2025
*  Purpose               : Method to split filename into name & path
*  RICEFW ID-Description : Common Utility Class Build
*----------------------------------------------------------------------*
* MODIFICATION LOG                                                     *
*----------------------------------------------------------------------*
*Date           Dev ID     TR#          Description                    *
*----------------------------------------------------------------------*
*DD-MMM-YYYY    SAPID      XXXXXXXXX    Change description             *
************************************************************************
    DATA: lv_len1      TYPE i,
          lv_len2      TYPE i,
          lv_len4      TYPE i,
          lv_tmpname   TYPE string,
          lv_tmpchar   TYPE abap_char1,
          lv_tmpcount  TYPE i,
          lv_lastslash TYPE i,
          lv_slash     TYPE abap_char1,
          lv_lastdot   TYPE i,
          lv_fullname  TYPE string,
          lv_filename  TYPE string.

    CLEAR: ev_path,ev_name.
    CHECK NOT iv_filename IS INITIAL.

    lv_filename = iv_filename.
    lv_len1      = strlen( lv_filename ).
    lv_tmpname   = lv_filename.
    " get directory separator based on location
    CALL METHOD z2cl_tech_comm_utilities_cld=>get_dir_seperator
      EXPORTING
        iv_source    = iv_location
      IMPORTING
        ev_separator = lv_slash.

    DO lv_len1 TIMES.
      lv_tmpchar = lv_tmpname+lv_tmpcount(1).
      lv_tmpcount = lv_tmpcount + 1.

      IF lv_tmpchar EQ lv_slash.
        lv_lastslash = lv_tmpcount.
      ELSEIF lv_tmpchar EQ '.'.
        lv_lastdot   = lv_tmpcount.
      ENDIF.
    ENDDO.
* Split the file name
    lv_len2 = lv_len1 - lv_lastslash.
    IF lv_lastslash > 0.
      ev_path = lv_filename+0(lv_lastslash).
    ENDIF.
    lv_fullname = lv_filename+lv_lastslash(lv_len2).

    IF lv_lastdot > 0 AND lv_lastdot > lv_lastslash.

      lv_len4     = lv_len1 - lv_lastdot.
      lv_len4     = strlen( lv_fullname ) - lv_len4 - 1.
      lv_filename = lv_fullname+0(lv_len4).
    ELSE.
      lv_filename = lv_fullname.
    ENDIF.
    ev_name = lv_filename.
  ENDMETHOD.
